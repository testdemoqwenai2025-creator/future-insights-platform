    16→// Helper to create mock requests
    17→function createMockRequest(overrides: Record<string, any> = {}): NextRequest {
    18→  const headers = new Headers();
    19→  
    20→  if (overrides.authorization) {
    21→    headers.set('authorization', overrides.authorization);
    22→  }
    23→  if (overrides.contentType !== undefined) {
    24→    if (overrides.contentType) {
    25→      headers.set('content-type', overrides.contentType);
    26→    }
    27→  } else {
    28→    headers.set('content-type', 'application/json');
    29→  }
    30→
    31→  let url = overrides.url || 'http://localhost:3000/api/status';
    32→  if (overrides.searchParams) {
    33→    const params = new URLSearchParams(overrides.searchParams).toString();
    34→    url = `${url}?${params}`;
    35→  }
    36→  
    37→  return {
    38→    headers,
    39→    method: overrides.method || 'GET',
    40→    url,
    41→    json: () => Promise.resolve(overrides.body || {}),
    42→    ip: '127.0.0.1'
    43→  } as unknown as NextRequest;
    44→}
    45→
    46→// Helper to parse response
    47→async function getResponseBody(response: Response): Promise<any> {
    48→  const text = await response.text();
    49→  try {
    50→    return JSON.parse(text);
    51→  } catch {
    52→    return text;
    53→  }
    54→}
