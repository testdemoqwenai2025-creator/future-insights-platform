    11→// Format Bytes Utility (from storage/route.ts)
    12→// ==========================================
    13→function formatBytes(bytes: number): string {
    14→  const units = ['B', 'KB', 'MB', 'GB', 'TB', 'PB', 'EB'];
    15→  let unitIndex = 0;
    16→  let value = bytes;
    17→  
    18→  while (value >= 1024 && unitIndex < units.length - 1) {
    19→    value /= 1024;
    20→    unitIndex++;
    21→  }
    22→  
    23→  return `${value.toFixed(1)} ${units[unitIndex]}`;
    24→}

    30→  isValidEmail(email: string): boolean {
    31→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    32→    return emailRegex.test(email);
    33→  },
    34→  
    35→  isValidUrl(url: string): boolean {
    36→    try {
    37→      new URL(url);
    38→      return true;
    39→    } catch {
    40→      return false;
    41→    }
    42→  },
    43→  
    44→  isValidOrcidId(orcid: string): boolean {
    45→    // ORCID ID format: 0000-0002-1825-0097 (4 groups of 4 digits)
    46→    const orcidRegex = /^\d{4}-\d{4}-\d{4}-\d{4}$/;
    47→    return orcidRegex.test(orcid);
    48→  },
    49→  
    50→  isStrongPassword(password: string): { valid: boolean; errors: string[] } {
    51→    const errors: string[] = [];
    52→    
    53→    if (password.length < 8) errors.push('Minimum 8 characters');
    54→    if (!/[A-Z]/.test(password)) errors.push('One uppercase letter');
    55→    if (!/[a-z]/.test(password)) errors.push('One lowercase letter');
    56→    if (!/\d/.test(password)) errors.push('One number');
    57→    if (!/[^A-Za-z0-9]/.test(password)) errors.push('One special character');
    58→    
    59→    return { valid: errors.length === 0, errors };
    60→  },
    61→  
    62→  sanitizeString(str: string): string {
    63→    return str
    64→      .trim()
    65→      .replace(/[<>\"\'&]/g, '') // Remove potentially dangerous characters
    66→      .replace(/\s+/g, ' ') // Collapse whitespace
    67→      .substring(0, 1000); // Limit length
    68→  }

    71→// ==========================================
    72→// Date/Time Utilities
    73→// ==========================================
    74→const dateUtils = {
    75→  formatDate(date: Date | string, format: 'iso' | 'short' | 'long' = 'iso'): string {
    76→    const d = typeof date === 'string' ? new Date(date) : date;
    77→    
    78→    switch (format) {
    79→      case 'short':
    80→        return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    81→      case 'long':
    82→        return d.toLocaleDateString('en-US', { 
    83→          weekday: 'long', 
    84→          year: 'numeric', 
    85→          month: 'long', 
    86→          day: 'numeric' 
    87→        });
    88→      default:
    89→        return d.toISOString();
    90→    }
    91→  },
    92→  
    93→  getRelativeTime(date: Date | string): string {
    94→    const d = typeof date === 'string' ? new Date(date) : date;
    95→    const now = new Date();
    96→    const diffMs = now.getTime() - d.getTime();
    97→    const diffSecs = Math.floor(diffMs / 1000);
    98→    const diffMins = Math.floor(diffSecs / 60);
    99→    const diffHours = Math.floor(diffMins / 60);
   100→    const diffDays = Math.floor(diffHours / 24);
   101→    
   102→    if (diffSecs < 60) return `${diffSecs}s ago`;
   103→    if (diffMins < 60) return `${diffMins}m ago`;
   104→    if (diffHours < 24) return `${diffHours}h ago`;
   105→    if (diffDays < 30) return `${diffDays}d ago`;
   106→    
   107→    return dateUtils.formatDate(d, 'short');
   108→  },
   109→  
   110→  isValidDate(date: any): boolean {
   111→    if (!(date instanceof Date) && typeof date !== 'string') return false;
   112→    const d = new Date(date);
   113→    return !isNaN(d.getTime());
   114→  }

   120→const stringUtils = {
   121→  truncate(str: string, maxLength: number, suffix: string = '...'): string {
   122→    if (str.length <= maxLength) return str;
   123→    return str.substring(0, maxLength - suffix.length) + suffix;
   124→  },
   125→  
   126→  capitalize(str: string): string {
   127→    return str.charAt(0).toUpperCase() + str.slice(1).toLowerCase();
   128→  },
   129→  
   130→  camelToTitle(str: string): string {
   131→    return str
   132→      .replace(/([A-Z])/g, ' $1')
   133→      .replace(/^./, (char) => char.toUpperCase())
   134→      .trim();
   135→  },
   136→  
   137→  slugify(str: string): string {
   138→    return str
   139→      .toLowerCase()
   140→      .trim()
   141→      .replace(/[^\w\s-]/g, '')
   142→      .replace(/[\s_]+/g, '-')
   143→      .replace(/^-+|-+$/g, '');
   144→  },
   145→  
   146→  generateId(prefix: string = ''): string {
   147→    const timestamp = Date.now().toString(36);
   148→    const random = Math.random().toString(36).substring(2, 8);
   149→    return prefix ? `${prefix}_${timestamp}${random}` : `${timestamp}${random}`;
   150→  },
   151→  
   152→  escapeHtml(str: string): string {
   153→    const htmlEntities: Record<string, string> = {
   154→      '&': '&amp;',
   155→      '<': '&lt;',
   156→      '>': '&gt;',
   157→      '"': '&quot;',
   158→      "'": '&#39;'
   159→    };
   160→    return str.replace(/[&<>"']/g, (char) => htmlEntities[char]);
   161→  }

   164→// ==========================================
   165→// Array/Object Utilities
   166→// ==========================================
   167→const collectionUtils = {
   168→  chunk<T>(array: T[], size: number): T[][] {
   169→    const chunks: T[][] = [];
   170→    for (let i = 0; i < array.length; i += size) {
   171→      chunks.push(array.slice(i, i + size));
   172→    }
   173→    return chunks;
   174→  },
   175→  
   176→  groupBy<T>(array: T[], keyFn: (item: T) => string): Record<string, T[]> {
   177→    return array.reduce((groups, item) => {
   178→      const key = keyFn(item);
   179→      if (!groups[key]) groups[key] = [];
   180→      groups[key].push(item);
   181→      return groups;
   182→    }, {} as Record<string, T[]>);
   183→  },
   184→  
   185→  unique<T>(array: T[]): T[] {
   186→    return [...new Set(array)];
   187→  },
   188→  
   189→  pick<T extends object, K extends keyof T>(obj: T, keys: K[]): Pick<T, K> {
   190→    const result = {} as Pick<T, K>;
   191→    keys.forEach(key => {
   192→      if (key in obj) {
   193→        result[key] = obj[key];
   194→      }
   195→    });
   196→    return result;
   197→  },
   198→  
   199→  omit<T extends object, K extends keyof T>(obj: T, keys: K[]): Omit<T, K> {
   200→    const result = { ...obj };
   201→    keys.forEach(key => {
   202→      delete result[key];
   203→    });
   204→    return result;
   205→  }
