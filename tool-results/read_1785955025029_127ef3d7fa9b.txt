    33→// Mock Next.js request
    34→function createMockRequest(overrides: Record<string, any> = {}): NextRequest {
    35→  const defaultHeaders = new Headers();
    36→  if (overrides.authorization) {
    37→    defaultHeaders.set('authorization', overrides.authorization);
    38→  }
    39→  if (overrides.origin) {
    40→    defaultHeaders.set('origin', overrides.origin);
    41→  }
    42→  if (overrides['X-API-Key']) {
    43→    defaultHeaders.set('X-API-Key', overrides['X-API-Key']);
    44→  }
    45→  if (overrides['X-Request-ID']) {
    46→    defaultHeaders.set('X-Request-ID', overrides['X-Request-ID']);
    47→  }
    48→  
    49→  return {
    50→    headers: defaultHeaders,
    51→    method: overrides.method || 'GET',
    52→    url: overrides.url || 'http://localhost:3000/api/test',
    53→    ip: overrides.ip || '127.0.0.1',
    54→    cookies: {
    55→      get: (name: string) => overrides.cookies?.[name] || null
    56→    }
    57→  } as unknown as NextRequest;
    58→}

   409→      rateLimit('unique_user_2', 5, 60000); // 3rd request - remaining: 2
   410→      
   411→      const result = rateLimit('unique_user_2', 5, 60000); // 4th request - remaining: 1

   415→    });

   417→    it('should block requests exceeding limit', () => {

   422→
   423→      const result = rateLimit('unique_user_3', 5, 60000);

   428→
   429→    it('should reset after window expires (simulated)', () => {

   785→  // Helper to extract JSON from NextResponse (handles ReadableStream)
   786→  async function getResponseBody(response: Response): Promise<any> {
   787→    if (!response.body) return {};
   788→    const text = await response.text();
   789→    try {
   790→      return JSON.parse(text);
   791→    } catch {
   792→      return text;
   793→    }
   794→  }
