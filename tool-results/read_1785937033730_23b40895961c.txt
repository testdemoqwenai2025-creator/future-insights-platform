   456→export default function AethelgardLandingPage() {
   457→  const [sidebarOpen, setSidebarOpen] = useState(true);
   458→  const [selectedFile, setSelectedFile] = useState<FileNode | null>(null);
   459→  const [expandedFolders, setExpandedFolders] = useState<Set<string>>(new Set(['application']));
   460→  const [activeTab, setActiveTab] = useState<string>('frontend');
   461→  const [currentTheme, setCurrentTheme] = useState<ThemeMode>('cosmic');
   462→  const [selectedAIMode, setSelectedAIMode] = useState<string>('hypothesis');
   463→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   464→  const [copiedPath, setCopiedPath] = useState<string | null>(null);
   465→
   466→  const theme = themes[currentTheme];
   467→
   468→  // Particle animation state
   469→  const [particles, setParticles] = useState<Array<{id: number; x: number; y: number; size: number; duration: number; delay: number}>>([]);
   470→
   471→  useEffect(() => {
   472→    const newParticles = Array.from({ length: 50 }, (_, i) => ({
   473→      id: i,
   474→      x: Math.random() * 100,
   475→      y: Math.random() * 100,
   476→      size: Math.random() * 4 + 1,
   477→      duration: Math.random() * 20 + 10,
   478→      delay: Math.random() * 5
   479→    }));
   480→    setParticles(newParticles);
   481→  }, []);
   482→
   483→  const toggleFolder = useCallback((path: string) => {
   484→    setExpandedFolders(prev => {
   485→      const next = new Set(prev);
   486→      if (next.has(path)) {
   487→        next.delete(path);
   488→      } else {
   489→        next.add(path);
   490→      }
   491→      return next;
   492→    });
   493→  }, []);
   494→
   495→  const copyFilePath = useCallback((path: string) => {
   496→    navigator.clipboard.writeText(path);
   497→    setCopiedPath(path);
   498→    setTimeout(() => setCopiedPath(null), 2000);
   499→  }, []);
   500→
   501→  const renderFileTree = useCallback((nodes: FileNode[], depth = 0) => {
   502→    return nodes.map((node) => {
   503→      const isExpanded = expandedFolders.has(node.path);
   504→      const isSelected = selectedFile?.path === node.path;
   505→
   506→      if (node.type === 'folder') {
   507→        return (
   508→          <div key={node.path}>
   509→            <button
   510→              onClick={() => toggleFolder(node.path)}
   511→              className="flex items-center gap-1.5 w-full px-2 py-1 text-left text-xs hover:bg-slate-800/50 rounded transition-colors"
   512→              style={{ paddingLeft: `${depth * 12 + 8}px` }}
   513→            >
   514→              {isExpanded ? (
   515→                <ChevronDown className="w-3.5 h-3.5 text-slate-400 flex-shrink-0" />
   516→              ) : (
   517→                <ChevronRight className="w-3.5 h-3.5 text-slate-400 flex-shrink-0" />
   518→              )}
   519→              {isExpanded ? (
   520→                <FolderOpen className="w-4 h-4 text-amber-400 flex-shrink-0" />
   521→              ) : (
   522→                <Folder className="w-4 h-4 text-amber-400/60 flex-shrink-0" />
   523→              )}
   524→              <span className={`truncate ${isSelected ? 'text-white' : 'text-slate-300'}`}>{node.name}</span>
   525→            </button>
   526→            {isExpanded && node.children && (
   527→              <div>{renderFileTree(node.children, depth + 1)}</div>
   528→            )}
   529→          </div>
   530→        );
   531→      }
   532→
   533→      return (
   534→        <button
   535→          key={node.path}
   536→          onClick={() => setSelectedFile(node)}
   537→          className="flex items-center gap-1.5 w-full px-2 py-1 text-left text-xs hover:bg-slate-800/50 rounded transition-colors group"
   538→          style={{ paddingLeft: `${depth * 12 + 20}px` }}
   539→        >
   540→          <FileCode className="w-4 h-4 text-violet-400 flex-shrink-0" />
   541→          <span className={`truncate flex-1 ${isSelected ? 'text-white' : 'text-slate-400'}`}>{node.name}</span>
   542→          <Copy
   543→            onClick={(e) => { e.stopPropagation(); copyFilePath(node.path); }}
   544→            className="w-3 h-3 opacity-0 group-hover:opacity-100 text-slate-500 hover:text-white transition-opacity flex-shrink-0"
   545→          />
   546→        </button>
   547→      );
   548→    });
   549→  }, [expandedFolders, selectedFile, toggleFolder, copyFilePath]);
   550→
   551→  return (
   552→    <div className={`min-h-screen ${theme.bgGradient} text-white overflow-x-hidden`}>
   553→      {/* Animated Background Particles */}
   554→      <div className="fixed inset-0 pointer-events-none overflow-hidden">
   555→        {particles.map((particle) => (
   556→          <div
   557→            key={particle.id}
   558→            className="absolute rounded-full animate-pulse"
   559→            style={{
   560→              left: `${particle.x}%`,
   561→              top: `${particle.y}%`,
   562→              width: `${particle.size}px`,
