   129→const TEMPLATE_SLUGS: Record<string, string> = {
   130→  'blast': 'blast-sequence-analysis',
   131→  'docking': 'molecular-docking-workflow',
   132→  'training': 'transformer-training-pipeline',
   133→  'bayesian': 'bayesian-inference-framework',

   138→
   139→function parseHashRoute(hash: string): { templateId: string | null; category: TemplateCategory | 'all' } {
   140→  const cleanHash = hash.replace(/^#\/?/, '');
   141→  const templatesMatch = cleanHash.match(/templates\/(?:([^/]+)\/)?([^/]+)\/?$/);
   142→  
   143→  if (templatesMatch) {
   144→    const categorySlug = templatesMatch[1];
   145→    const templateSlug = templatesMatch[2];
   146→    
   147→    // Direct ID match
   148→    const directMatch = templates.find(t => t.id === templateSlug);
   149→    if (directMatch) return { templateId: directMatch.id, category: directMatch.category };
   150→    
   151→    // Slug mapping
   152→    const mappedId = TEMPLATE_SLUGS[templateSlug.toLowerCase()];
   153→    if (mappedId) {
   154→      const template = templates.find(t => t.id === mappedId);
   155→      if (template) return { templateId: template.id, category: template.category };
   156→    }
   157→    
   158→    // Category only
   159→    if (categorySlug) {
   160→      const mappedCategory = CATEGORY_SLUGS[categorySlug.toLowerCase()];
   161→      if (mappedCategory) return { templateId: null, category: mappedCategory };
   162→    }
   163→  }
   164→  
   165→  return { templateId: null, category: 'all' };
   166→}
