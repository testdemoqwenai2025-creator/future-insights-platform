    14→// Helper to create mock requests
    15→function createMockRequest(overrides: Record<string, any> = {}): NextRequest {
    16→  const headers = new Headers();
    17→  
    18→  if (overrides.authorization) {
    19→    headers.set('authorization', overrides.authorization);
    20→  }
    21→  if (overrides.contentType !== undefined) {
    22→    if (overrides.contentType) {
    23→      headers.set('content-type', overrides.contentType);
    24→    }
    25→  } else {
    26→    headers.set('content-type', 'application/json');
    27→  }
    28→
    29→  let url = overrides.url || 'http://localhost:3000/api/papers';
    30→  if (overrides.searchParams) {
    31→    const params = new URLSearchParams(overrides.searchParams).toString();
    32→    url = `${url}?${params}`;
    33→  }
    34→  
    35→  return {
    36→    headers,
    37→    method: overrides.method || 'GET',
    38→    url,
    39→    json: () => Promise.resolve(overrides.body || {}),
    40→    ip: '127.0.0.1'
    41→  } as unknown as NextRequest;
    42→}
    43→
    44→// Helper to parse response
    45→async function getResponseBody(response: Response): Promise<any> {
    46→  const text = await response.text();
    47→  try {
    48→    return JSON.parse(text);
    49→  } catch {
    50→    return text;
    51→  }
    52→}
