    34→    34→function createMockRequest(overrides: Record<string, any> = {}): NextRequest {
    35→    35→  const defaultHeaders = new Headers();
    36→    36→  if (overrides.authorization) {
    37→    37→    defaultHeaders.set('authorization', overrides.authorization);
    38→    38→  }
    39→    39→  if (overrides.origin) {
    40→    40→    defaultHeaders.set('origin', overrides.origin);
    41→    41→  }
    42→    42→  if (overrides['X-API-Key']) {
    43→    43→    defaultHeaders.set('X-API-Key', overrides['X-API-Key']);
    44→    44→  }
    45→    45→  if (overrides['X-Request-ID']) {
    46→    46→    defaultHeaders.set('X-Request-ID', overrides['X-Request-ID']);
    47→    47→  }
    48→    48→  
    49→    49→  return {
    50→    50→    headers: defaultHeaders,
    51→    51→    method: overrides.method || 'GET',
    52→    52→    url: overrides.url || 'http://localhost:3000/api/test',
    53→    53→    ip: overrides.ip || '127.0.0.1',
    54→    54→    cookies: {
    55→    55→      get: (name: string) => overrides.cookies?.[name] || null
    56→    56→    }
    57→    57→  } as unknown as NextRequest;
    58→    58→}
